Unity Shader（9）：常用效果实战——顶点动画、溶解与描边

Created2026-08-13|Updated2026-08-17|UnityShader
|Post Views:

前面的语法学得差不多了，这一讲把它们串起来，动手做三个立刻能用的效果：顶点动画（波浪）、溶解（噪声裁剪）、描边（法线外扩壳）。每个效果都给出完整代码和参数说明，并点出常见坑。做完这三个，你就具备了自己写”特效型 Shader”的基本手感。

一、顶点动画：让模型”活”起来

1.1 原理

顶点动画的关键在于：在 vert 里移动顶点的位置。因为 vert 每帧、每个顶点都会执行，所以只要把位置写成时间的函数，模型就会动起来。

最常见的是正弦波浪——用顶点的某个坐标做相位，用 _Time.y 做时间驱动：

float wave = sin(v.vertex.x * _Frequency + _Time.y * _Speed) * _Amplitude;
v.vertex.y += wave;
```text

### 1.2 完整代码

```hlsl
Shader "Tutorial/VertexWave"
{
    Properties
    {
        _MainTex ("Texture", 2D) = "white" {}
        _Amplitude ("Amplitude", Range(0, 1)) = 0.1
        _Frequency ("Frequency", Float) = 1
        _Speed ("Speed", Float) = 1
    }

    SubShader
    {
        Tags { "RenderType"="Opaque" }

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "UnityCG.cginc"

            sampler2D _MainTex;
            float4 _MainTex_ST;
            float _Amplitude, _Frequency, _Speed;

            struct appdata { float4 vertex : POSITION; float2 uv : TEXCOORD0; };
            struct v2f     { float4 vertex : SV_POSITION; float2 uv : TEXCOORD0; };

            v2f vert (appdata v)
            {
                v2f o;
                float wave = sin(v.vertex.x * _Frequency + _Time.y * _Speed) * _Amplitude;
                v.vertex.y += wave;
                o.vertex = UnityObjectToClipPos(v.vertex);
                o.uv = TRANSFORM_TEX(v.uv, _MainTex);
                return o;
            }

            fixed4 frag (v2f i) : SV_Target
            {
                return tex2D(_MainTex, i.uv);
            }
            ENDCG
        }
    }
}
```text

### 1.3 常见坑

- **法线不会跟着动**：顶点位置变了，但法线还是原来的，光照会显得"没变化"。真实水面/布料要么重算法线，要么用贴图遮一下，或者接受"无光照"效果。
- **阴影不会跟着动**：顶点动画发生在 Shader 里，阴影 Pass 用的是原始顶点。要让阴影也跟着动，需要自定义阴影投射 Pass（高级话题）。
- 顶点动画适合**顶点数量足够的网格**，低模上波浪会很"块状"。

## 二、溶解：噪声 + clip 裁剪

### 2.1 原理

溶解的核心是 `clip()`：当传入值小于 0 时，丢弃这个片元。我们拿一张噪声图的值减去一个阈值，把"低于阈值"的部分裁掉：

```hlsl
float noise = tex2D(_NoiseTex, uv).r;
clip(noise - _Amount);   // _Amount 从 0 到 1，逐渐裁掉更多
```text

再配合 `smoothstep`，可以在溶解边缘加一圈发光色：

```hlsl
float edge = smoothstep(_Amount, _Amount + _EdgeWidth, noise);
col.rgb = lerp(_EdgeColor.rgb, col.rgb, edge);
```text

### 2.2 完整代码

```hlsl
Shader "Tutorial/Dissolve"
{
    Properties
    {
        _MainTex ("Texture", 2D) = "white" {}
        _NoiseTex ("Noise", 2D) = "white" {}
        _Amount ("Dissolve", Range(0, 1)) = 0
        _EdgeColor ("Edge Color", Color) = (1, 0.4, 0, 1)
        _EdgeWidth ("Edge Width", Range(0, 0.2)) = 0.05
    }

    SubShader
    {
        Tags { "RenderType"="Opaque" }

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "UnityCG.cginc"

            sampler2D _MainTex;
            float4 _MainTex_ST;
            sampler2D _NoiseTex;
            float4 _NoiseTex_ST;
            float _Amount, _EdgeWidth;
            fixed4 _EdgeColor;

            struct appdata { float4 vertex : POSITION; float2 uv : TEXCOORD0; };
            struct v2f
            {
                float4 vertex : SV_POSITION;
                float2 uv : TEXCOORD0;
                float2 uvNoise : TEXCOORD1;
            };

            v2f vert (appdata v)
            {
                v2f o;
                o.vertex = UnityObjectToClipPos(v.vertex);
                o.uv = TRANSFORM_TEX(v.uv, _MainTex);
                o.uvNoise = TRANSFORM_TEX(v.uv, _NoiseTex);
                return o;
            }

            fixed4 frag (v2f i) : SV_Target
            {
                fixed4 col = tex2D(_MainTex, i.uv);
                float noise = tex2D(_NoiseTex, i.uvNoise).r;

                clip(noise - _Amount);

                float edge = smoothstep(_Amount, _Amount + _EdgeWidth, noise);
                col.rgb = lerp(_EdgeColor.rgb, col.rgb, edge);
                return col;
            }
            ENDCG
        }
    }
}
```text

### 2.3 常见坑

- 噪声图要选**灰度噪声**（如 Perlin/Voronoi），否则用 `.r` 通道取出的值不连续，溶解边缘会乱。
- `clip` 丢弃的片元边缘是"硬"的，想软化就调大 `_EdgeWidth` 或用抗锯齿。
- 溶解物体通常还需要在 C# 侧配合一个随时间增长的 `_Amount` 动画。

## 三、描边：法线外扩（Shell 方法）

### 3.1 原理

描边最常见的做法是"背面壳"：**再画一个 Pass，把顶点沿法线方向往外扩一点，只显示背面，涂成描边色**。因为壳比原模型略大，露出的边缘就成了描边。

```text
Pass 1：正常画模型
Pass 2：Cull Front（只画背面）+ 顶点沿法线外扩 + 纯色
```text

### 3.2 完整代码

```hlsl
Shader "Tutorial/Outline"
{
    Properties
    {
        _MainTex ("Texture", 2D) = "white" {}
        _OutlineColor ("Outline Color", Color) = (0, 0, 0, 1)
        _OutlineWidth ("Outline Width", Range(0, 0.1)) = 0.02
    }

    SubShader
    {
        Tags { "RenderType"="Opaque" }

        // Pass 0：正常贴图
        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "UnityCG.cginc"
            sampler2D _MainTex; float4 _MainTex_ST;
            struct appdata { float4 vertex : POSITION; float2 uv : TEXCOORD0; };
            struct v2f     { float4 vertex : SV_POSITION; float2 uv : TEXCOORD0; };
            v2f vert (appdata v) { v2f o; o.vertex = UnityObjectToClipPos(v.vertex); o.uv = TRANSFORM_TEX(v.uv, _MainTex); return o; }
            fixed4 frag (v2f i) : SV_Target { return tex2D(_MainTex, i.uv); }
            ENDCG
        }

        // Pass 1：描边壳
        Pass
        {
            Cull Front
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "UnityCG.cginc"
            fixed4 _OutlineColor;
            float _OutlineWidth;
            struct appdata { float4 vertex : POSITION; float3 normal : NORMAL; };
            struct v2f     { float4 vertex : SV_POSITION; };
            v2f vert (appdata v)
            {
                v2f o;
                v.vertex.xyz += normalize(v.normal) * _OutlineWidth;
                o.vertex = UnityObjectToClipPos(v.vertex);
                return o;
            }
            fixed4 frag (v2f i) : SV_Target { return _OutlineColor; }
            ENDCG
        }
    }
}
```text

### 3.3 常见坑

- **硬边网格会断裂**：沿顶点法线外扩要求"相邻面共享平滑法线"，硬边（不同法线）会在棱角处裂开。解决办法是烘焙平滑法线、或用顶点色控制描边宽度、或改用屏幕空间后处理描边。
- **近大远小**：固定世界单位外扩，远处描边会显得细。想做"屏幕恒定宽度"要用后处理描边。
- **法线外扩 vs 屏幕空间**：Shell 方法简单直观，但不稳定；**屏幕空间描边**（基于深度/法线做边缘检测）更稳定，属于后处理范畴——下一讲会讲。
